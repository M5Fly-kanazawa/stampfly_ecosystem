\begin{frame}{ドローンは「勝手に」飛んでいるのではない}
  \begin{block}{Tello はなぜ「言うことを聞く」のか}
    \small 「前に進め」で前に進み、「回れ」で回る --- それは膨大な\textbf{試行錯誤（制御工学）の賜物}です。\\
    素のドローンは、制御なしでは\textbf{1秒も浮いていられません}
  \end{block}

  \vspace{0.3em}

  \begin{itemize}\small
    \item クアッドコプタは本質的に不安定 --- 放っておくと勝手に傾いて墜落する
    \item センサで姿勢を測り、4つのモータを絶えず調整し続けて、はじめて浮いていられる
  \end{itemize}

  \vspace{0.3em}

  \begin{exampleblock}{この講座の目標}
    \small \textbf{「飛ばないドローン」を制御して飛ばせるエンジニアを育てる} --- StampFly はそのための教材
  \end{exampleblock}
\end{frame}

\begin{frame}{安全ルール：飛行の安全}
  \begin{block}{飛行前の確認}
    \begin{itemize}\small\setlength{\itemsep}{0pt}
      \item 保護メガネを全員着用
      \item 飛行エリアは区画・ネットで分離
      \item プロペラガードの状態を確認
      \item 長い髪はまとめる
      \item 機体の真上に顔を出さない
    \end{itemize}
  \end{block}

  \vspace{0.15em}

  \begin{alertblock}{緊急停止}
    \footnotesize \warn{DISARM} = モータを止める操作。飛行中は自動着陸 → もう一度で即時停止
  \end{alertblock}

  \vspace{0.1em}

  \begin{exampleblock}{バッテリー}
    \footnotesize 満充電 4.2\,V／3.4\,V で低電圧警告。膨らんだ電池は使用しない。充電中は目を離さない
  \end{exampleblock}
\end{frame}
