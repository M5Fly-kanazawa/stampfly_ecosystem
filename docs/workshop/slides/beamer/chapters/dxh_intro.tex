\begin{frame}{安全ルール（1）：飛行の安全}
  \begin{block}{飛行前の確認}
    \begin{itemize}\small\setlength{\itemsep}{0pt}
      \item 保護メガネを全員着用
      \item 飛行エリアは区画・ネットで分離
      \item プロペラガードの状態を確認
      \item 長い髪はまとめる
      \item 機体の真上に顔を出さない
    \end{itemize}
  \end{block}

  \vspace{0.15em}

  \begin{alertblock}{緊急停止}
    \footnotesize \warn{DISARM}（自動着陸が始まる）→ 着陸中にもう一度 \warn{DISARM}（即時モータ停止）
  \end{alertblock}

  \vspace{0.1em}

  \begin{exampleblock}{バッテリー}
    \footnotesize 満充電 4.2\,V／3.4\,V で低電圧警告。膨らんだ電池は使用しない。充電中は目を離さない
  \end{exampleblock}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{安全ルール（2）：モータ実習の安全}
  \begin{block}{プロペラを回す実習ならではの注意}
    \begin{itemize}\small
      \item \warn{Duty は 0.15 以下}（ファームでは強制されないので数値を必ずダブルチェック）
      \item プロペラは付けたまま --- 機体上部の基板を上から\textbf{指で押さえて}回す
      \item バッテリーと USB の同時接続は避ける（動作確認はバッテリー駆動）
    \end{itemize}
  \end{block}

  \vspace{0.3em}

  \begin{alertblock}{止め方}
    \footnotesize 機体本体ボタンをもう一度クリック（DISARM）／完全に切るときはバッテリーを外す
  \end{alertblock}
\end{frame}
