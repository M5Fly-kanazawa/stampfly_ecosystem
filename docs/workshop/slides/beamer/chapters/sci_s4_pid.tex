\begin{frame}{要点3つ / Three Key Points}
  \begin{enumerate}
    \item モータ+機体のプラント伝達関数 $G_p(s) = K/(s(\tau_m s+1))$ を実測パラメータから導出し，ゲイン設計に使う
    \item P 制御から PID 化し，不完全微分・アンチワインドアップ・積分リセットまで含めたvehicle 実装と学習者コードを対比する
    \item フライトデータからプラントを同定し，設計値と実測値を突き合わせる
  \end{enumerate}

  \vspace{0.5em}

  \begin{alertblock}{本セッションの立ち位置}
    \small
    ここまでのレッスンは動かすことが目的だった。ここからは，動いた結果を
    モデルと数値で説明できるかを問う

\begin{frame}{2次系設計 / Second-Order Design}
  \vspace{-2mm}
  \begin{columns}[c]
    \begin{column}{0.42\textwidth}
      \centering
      \resizebox{\columnwidth}{!}{\input{../tikz/step_response_zeta}}
    \end{column}
    \begin{column}{0.58\textwidth}
      \begin{block}{閉ループ伝達関数（2次系）}
        \vspace{-0.3em}
        $G_{cl}(s) = \dfrac{K_p K}{\tau_m s^2 + s + K_p K}$
        \vspace{-0.7em}
      \end{block}

      \vspace{0.2em}

      \begin{block}{$\zeta$ から $K_p$ を逆算する設計式}
        \vspace{-0.3em}
        \[
          K_p = \frac{1}{4\zeta^2 K \tau_m}
        \]
        \vspace{-0.7em}
      \end{block}

      \vspace{0.2em}

      {\small
      Roll/Pitch で $\zeta = 0.7$（オーバーシュート約5\%）を狙うと
      $K_{p,roll}=0.25$，$K_{p,pitch}=0.36$}
    \end{column}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{開ループボード線図 / Open-Loop Bode Plot}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \centering
      \resizebox{\columnwidth}{!}{\input{../tikz/bode_loop_shaping}}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{block}{開ループ伝達関数}\small
        \vspace{-0.3em}
        \[
          L(s) = K_p \cdot G_p(s) = \frac{K_p K}{s(\tau_m s + 1)}
        \]
        \vspace{-0.7em}
      \end{block}

      \vspace{0.3em}

      {\small $K_p$ を上げるとゲイン交差周波数 $\omega_{gc}$ が上がり，位相余裕（PM）は下がる。
      P 制御だけでは応答速度と安定性がトレードオフの関係にある}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{無駄時間の影響 / Dead Time Effect}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \resizebox{\columnwidth}{!}{\input{../tikz/bode_dead_time}}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{block}{制御ループの無駄時間}\small
        $\tau_d \approx 5$\,ms（センサ処理 + 制御演算 + PWM 更新）
        \vspace{-0.3em}
        \[
          L_{delay}(s) = L(s)\cdot e^{-\tau_d s}
        \]
        \vspace{-1em}
      \end{block}

      \vspace{0.3em}

      \begin{alertblock}{\small 実用上 PM $\geq 60°$ が必要}\scriptsize
        モデル上は安定なゲインでも，無駄時間を足すと実機で振動することがある
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{PID 工学形式パラメータ / Engineering Form}
  \begin{table}
    \centering\scriptsize
    \renewcommand{\arraystretch}{0.95}
    \begin{tabular}{llp{58mm}}
      \hline
      \textbf{パラメータ} & \textbf{意味} & \textbf{効果} \\
      \hline
      $K_p$  & 比例ゲイン   & 大きいほど応答が速い，大きすぎると振動 \\
      $T_i$  & 積分時間 [s] & 小さいほど積分が強く効き偏差が早く消える \\
      $T_d$  & 微分時間 [s] & 大きいほど微分が強く効き振動を抑える \\
      $\eta$ & フィルタ係数  & $0.1\sim0.2$，大きいほどノイズに強い \\
      \hline
    \end{tabular}
  \end{table}

  \begin{block}{教科書形式との対応}\footnotesize
    $K_i = K_p / T_i\,,\quad K_d = K_p \cdot T_d$
  \end{block}

  {\footnotesize L8 の解答が使うゲイン $(K_p, T_i, T_d)$（$\eta=0.125$ 共通）:
  Roll $(0.25,\,1.67,\,0.01)$，Pitch $(0.36,\,1.67,\,0.01)$。$K_p$ は前ページの $\zeta=0.7$ 設計値}
  \par\vspace{0.2em}
  {\scriptsize Yaw は L8 の解答では経験値 $K_p=2.0$ を使う。同じ設計式を最新の実測値 $K=8.0$ に
  適用すると $K_p=3.19$ になり，設計式で追い直せる例として挙げる}
\end{frame}

\begin{frame}[fragile]{D 項とアンチワインドアップ / D-Term \& Anti-Windup}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{L8 学習者コード}\scriptsize
        理想 PID から不完全微分フィルタへ置換
      \end{block}
      \begin{lstlisting}[style=sfcpp, basicstyle=\ttfamily\scriptsize]
float D = Kp*Td*(err-prev_err)/dt;
// -> incomplete derivative filter
float alpha = 2*eta*Td/dt;
float a=(alpha-1)/(alpha+1);
float b=2*Td/((alpha+1)*dt);
d_filt = a*d_filt + b*(err-prev_err);
      \end{lstlisting}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{vehicle 実装: sf\_controller\_pid}\scriptsize
        微分は誤差ではなく\textbf{測定値}に掛ける（D-on-M）。目標値が
        階段状に変わっても微分キックが出ない
      \end{alertblock}
      \begin{exampleblock}{アンチワインドアップ}\scriptsize
        単純な積分クランプではなく，出力が飽和方向に押される間だけ
        積分を止める「条件付き積分」を使う
      \end{exampleblock}
    \end{column}
  \end{columns}

  {\scriptsize L8 では振動と定常偏差の体感を優先し，この2点は扱わない。
  \code{sf\_controller\_pid/include/pid.hpp} が vehicle 実装のコード}
\end{frame}

\begin{frame}{離散化と ARM 時リセット / Discretization \& Reset}
  \begin{block}{離散化: 双一次変換（Tustin）}\small
    比例・積分・微分すべてに双一次変換を使う。L8 の不完全微分フィルタの式
    （$\alpha, a, b$）はこの変換そのもの
  \end{block}

  \vspace{0.3em}

  \begin{exampleblock}{積分リセットのタイミング}\small
    積分器は DISARM 中ずっとゼロなのではなく，\textbf{ARM 遷移のたびに}
    リセットされる。着地して DISARM し，次に ARM した瞬間に積分が空になり，
    前のフライトの偏差を引きずらない
  \end{exampleblock}

  \vspace{0.3em}

  {\footnotesize
  ライブチューニング（\code{param set} での即時反映）では，積分器の状態は
  \textbf{保持}される。ゲイン変更のたびにループを蹴らないための区別}
\end{frame}

\begin{frame}{姿勢推定: ジャイロドリフト問題 / Gyro Drift Problem}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \resizebox{\linewidth}{!}{\input{../tikz/gyro_drift}}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{itemize}\small
        \item ジャイロは角速度 $\omega$ を測定，角度は積分 $\theta=\int\omega\,dt$ が必要
        \item バイアスが蓄積し，ドリフトが増大する
      \end{itemize}
      \vspace{0.5em}
      \begin{alertblock}{問題}\small
        ジャイロ単体では長時間の姿勢推定ができない。STABILIZE 以上で
        姿勢角を目標にするには，角度そのものの推定が要る
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{相補フィルタと STABILIZE / Complementary Filter}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \centering
      \resizebox{\linewidth}{!}{\input{../tikz/comp_filter}}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{数式}\footnotesize
        $\hat\theta_k = \alpha(\hat\theta_{k-1}+\omega\Delta t) + (1-\alpha)\theta_{accel}$
      \end{block}
      \vspace{0.3em}
      \begin{itemize}\small
        \item $\alpha=0.98$（ジャイロ98\% + 加速度2\%）
        \item StampFly の既定は 15 状態 ESKF。相補フィルタは自作して

      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.3em}

  {\small STABILIZE ではこの推定角を外側ループの入力にし，スティックが
  傾き角そのものを指令する}
\end{frame}

\begin{frame}[fragile]{システム同定: sf sysid fit / Model Fitting}
  \begin{block}{$K_p$ が既知なら，閉ループデータから開ループプラントを復元できる}
    \footnotesize $u(t) = K_p(\text{target}-\text{gyro})$，$y(t)=\text{gyro}$ として
    $G_p(s)=K/(s(\tau_m s+1))$ に最小二乗フィット
  \end{block}

  \vspace{0.5em}

  \begin{sflisting}[basicstyle=\ttfamily\scriptsize]{出力例}
sf sysid fit flight.csv --kp 0.5 --plot
  Roll  K= 98.5 (ref:102.0, err:3.4%) tau_m=0.019 R2=0.94
  Pitch K= 72.1 (ref: 70.0, err:3.0%) tau_m=0.021 R2=0.91
  \end{sflisting}

  {\small 理論値（L6）との誤差3〜4\%は，モデル簡略化と無駄時間の影響で説明できる}
\end{frame}

\begin{frame}[fragile]{自動チューニング: sf sysid rate-fit / rate-tune}
  \begin{block}{ETFE によるプラント同定と仕様ベースのゲイン設計}\footnotesize
    チャープ（またはダブレット）励振で軸ごとに $(b, L, T)$ を推定し，
    $G_p(s) = b\,e^{-Ls}/(s(Ts+1))$ にフィットする。sf sysid fit と違い
    $K_p$ を仮定しない開ループ同定
  \end{block}

  \begin{sflisting}[basicstyle=\ttfamily\scriptsize]{コマンド}
sf sysid rate-fit flight.csv --axis roll -o fit.json
sf sysid rate-tune --fit fit.json --wc 25 --pm 60
  \end{sflisting}

  {\footnotesize \code{rate-tune} はゲイン交差周波数 $\omega_c$ と位相余裕 PM の仕様を満たす
  $K_p/T_i/T_d$ を逆算し，\code{param set rate.roll.kp ...} の形で出力する}

  {\scriptsize \warn{注:} 学習者コードで rate-fit/rate-tune を使うには，角速度目標を計算した
  直後に \api{ws::set\_rate\_target(roll,pitch,yaw)} を呼ぶ（L7 で追加，Data Stream の
  \code{rate\_ref\_*} に記録される）}
\end{frame}
