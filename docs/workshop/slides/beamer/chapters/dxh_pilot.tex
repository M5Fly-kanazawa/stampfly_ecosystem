\begin{frame}{①（参考）購入したコントローラのセットアップ}
  \begin{alertblock}{ご自身で購入したコントローラをお使いの場合}
    \footnotesize M5Atom Joystick は購入時 \textbf{M5Stack 社の出荷時ファーム}のままです。StampFly 用ではないため \textbf{最初に一度だけ}専用ファームウェアの書き込みが必要です
  \end{alertblock}

  \begin{block}{書き込み手順（PC 側の開発環境が構築済みであること）}
    \begin{enumerate}\footnotesize
      \setlength{\itemsep}{0pt}
      \item コントローラを USB ケーブルで PC に接続する
      \item \code{sf build controller}
      \item \code{sf flash controller}
    \end{enumerate}
  \end{block}

  \vspace{0.1em}

  \begin{exampleblock}{確認}
    \footnotesize LCD に StampFly 用の起動画面が出れば OK。次ページの USB HID モード切替に進めます
  \end{exampleblock}

  {\footnotesize ※ 講座当日お渡しした貸出コントローラは書き込み済みのため、この手順は不要でした}
\end{frame}

\begin{frame}{①（参考）ご自宅・学校でシミュレータを試すには}
  \begin{center}
    \resizebox{!}{0.28\textheight}{\input{../tikz/controller_annotated}}
  \end{center}

  \begin{block}{USB HID モードへの切替（初期状態 = ESP-NOW から）}
    \begin{enumerate}\footnotesize
      \setlength{\itemsep}{0pt}
      \item 上図の\textbf{LCD画面}（＝\textbf{M5ボタン}）を\textbf{短く1回}押す $\to$ メニューが開く
      \item \textbf{右スティック}を上下に倒し「Comm: ESP-NOW」にカーソルを合わせる
      \item \textbf{右ボタン}（STABILIZE/ACRO切替と共用）を押して選択 $\to$「Comm: UDP」に変わる（まだ使えない）
      \item \textbf{もう一度右ボタン}を押して選択 $\to$「Comm: USB HID」に切り替わり自動的に再起動
    \end{enumerate}
  \end{block}

  {\scriptsize ※ 今日お渡しする機材は事前設定済みのため不要な手順です。ご自身の機材の場合、USBケーブルは「データ通信対応」のものを使用してください（充電専用ケーブルでは認識されません）}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{① シミュレータ → 実機への切り替え}
  \begin{block}{コントローラを実機用（ESP-NOW）に戻す}
    \begin{enumerate}\small
      \item コントローラの \textbf{M5 ボタン}（画面下）でメニューを開く
      \item \textbf{「通信モード」を1回選ぶ} $\to$ USB HID から ESP-NOW（ルーターなしで機体と直接つながる無線）に切り替わり、自動で再起動する
      \item USB ケーブルを外す（実機とは電波でつながる）
    \end{enumerate}
  \end{block}

  \vspace{0.3em}

  {\footnotesize ※ 通信モードを切り替えても、ペアリングなどの設定はそのまま残ります}
\end{frame}

\begin{frame}{① 機体の起動と LED の見方}
  \begin{table}
    \centering\small
    \begin{tabular}{ll}
      \hline
      \textbf{LED・音} & \textbf{状態} \\
      \hline
      白色点灯 + 起動音 & 起動中 \\
      マゼンタ点滅 & ジャイロ校正中（機体に触らない） \\

      \hline
    \end{tabular}
  \end{table}

  \vspace{0.5em}

  \begin{alertblock}{起動時の注意}
    平らな床に置いて電源を入れる。校正が終わる（緑常灯）まで機体を動かさない
  \end{alertblock}
\end{frame}

\begin{frame}{① ALT\_HOLD と自動離着陸}
  \begin{block}{自動離着陸と ARM / DISARM}
    \small \textbf{ARM}（右スティックを押し込む）$\to$ 自動離陸・高度0.5\,m を自動維持\\
    \textbf{DISARM}（もう一度押し込む）$\to$ 自動着陸 $\to$ 着陸中にもう一度で緊急停止
  \end{block}

  \vspace{0.3em}

  \begin{alertblock}{低電圧警告}
    \footnotesize シアン点滅 + 警告音 $\to$ 着陸して電池交換（飛行時間 約4分）
  \end{alertblock}

  \vspace{0.3em}

  \begin{exampleblock}{進め方}
    \footnotesize 1人1セット・全員同時に飛行（交代なし）。ペアリングは1対1 --- 隣の機体は反応しません。電池が切れたら（シアン点滅）交換して再開
  \end{exampleblock}
\end{frame}
