\begin{frame}{① シミュレータ → 実機への切り替え}
  \begin{block}{コントローラを実機用（ESP-NOW）に戻す}
    \begin{enumerate}\small
      \item コントローラの \textbf{M5 ボタン}（画面下）でメニューを開く
      \item \textbf{「通信モード」を1回選ぶ} $\to$ USB HID から ESP-NOW に切り替わり、自動で再起動する
      \item USB ケーブルを外す（実機とは電波でつながる）
    \end{enumerate}
  \end{block}

  \vspace{0.3em}

  \begin{exampleblock}{安心してください}
    \footnotesize 切り替えてもペアリングなどの設定は消えません
  \end{exampleblock}
\end{frame}

\begin{frame}{① 機体の起動と LED の見方}
  \begin{table}
    \centering\small
    \begin{tabular}{ll}
      \hline
      \textbf{LED・音} & \textbf{状態} \\
      \hline
      白色点灯 + 起動音 & 起動中 \\
      マゼンタ点滅 & ジャイロ校正中（機体に触らない） \\
      緑色常灯 + ピッ×3 & 準備完了（約10秒） \\
      \hline
    \end{tabular}
  \end{table}

  \vspace{0.5em}

  \begin{alertblock}{起動時の注意}
    平らな床に置いて電源を入れる。校正が終わる（緑常灯）まで機体を動かさない
  \end{alertblock}
\end{frame}

  \end{exampleblock}

\begin{frame}{① ALT\_HOLD と自動離着陸}
  \begin{block}{自動離着陸と ARM / DISARM}
    \small \textbf{ARM}（右スティックを押し込む）$\to$ 自動離陸・高度0.5\,m を自動維持\\
    \textbf{DISARM}（もう一度押し込む）$\to$ 自動着陸 $\to$ 着陸中にもう一度で緊急停止
  \end{block}

  \vspace{0.3em}

  \begin{alertblock}{低電圧警告}
    \footnotesize シアン点滅 + 警告音 $\to$ 着陸して電池交換（飛行時間 約4分）
  \end{alertblock}

  \vspace{0.3em}

  \begin{exampleblock}{進め方}
    \footnotesize 1人1セットで全員同時に飛行（交代なし）。機体は事前ペアリング済みなので操縦だけに集中できます。電池が切れたら（シアン点滅）交換して再開
  \end{exampleblock}
\end{frame}
