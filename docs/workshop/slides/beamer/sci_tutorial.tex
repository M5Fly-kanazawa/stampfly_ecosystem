% --- SCI tutorial session-divider macro ------------------------------------
% \lessondivider (stampfly_slides.sty) hardcodes the word "Lesson N". The SCI
% tutorial is organised as five timetabled sessions, so define a local divider
% that shows the session number, title and time slot instead. The shared style
% file is left untouched (same approach as dxh_workshop.tex).
% \lessondivider は "Lesson N" を固定表示するため、時間割で区切る本チュートリアル
% には合わない。共有スタイルは触らず、セッション番号・題目・時間帯を出す専用の
% 区切りマクロをここで定義する（dxh_workshop.tex と同じ方針）。
\newcommand{\scisession}[3]{%
  \begin{frame}[plain]
    \centering
    \vspace*{\fill}
    \begin{tikzpicture}
      \node[
        rectangle, rounded corners=8pt,
        draw=sfblue, line width=1.5pt,
        fill=sflight,
        inner xsep=20pt, inner ysep=16pt,
        align=center,
        text width=120mm,
      ] {%
        {\fontsize{30}{36}\selectfont\bfseries\textcolor{sfblue}{Session #1}}\\[8pt]
        {\Large\bfseries\textcolor{sfdark}{#2}}\\[4pt]
        {\normalsize\textcolor{sfgray}{#3}}%
      };
    \end{tikzpicture}
    \vspace*{\fill}
    \par
    {\small\textcolor{sfgray}{SCI/SICE チュートリアル講座 2026}}\par
    \vspace{8mm}
  \end{frame}
}

% carries NO QR in either build (\sciTitleQR is empty on both branches below).
% The \ifdefined branch structure is kept for parity with the other decks so a
% QR can be dropped in later without restructuring.
% ビルド2系統についての注記（今回の修正で訂正）: dxh_workshop.tex は本番
% ビルドでタイトルページ右下に自分自身のドクセル公開ページへのQR
% （qr_docswell_dxh.png）を出している。本デッキにはそのQR画像がまだ存在しない
% （docs/workshop/slides/images/qr_docswell_sci* が無い）ため、タイトルページは
% どちらのビルドでもQRなし（下の \sciTitleQR は両分岐とも空）。\ifdefined
% 分岐の構造だけは他デッキと揃えて残してあり、後でQR画像を追加する際に
% 構造変更が不要になるようにしてある。
\ifdefined\DocswellBuild
  \newcommand{\sciTitleQR}{}
\else
  \newcommand{\sciTitleQR}{}
\fi
