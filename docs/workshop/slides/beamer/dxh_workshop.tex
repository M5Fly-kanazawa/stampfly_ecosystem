% --- DXH part-divider macro -------------------------------------------------
% \lessondivider (stampfly_slides.sty) hardcodes the word "Lesson N", which
% does not fit this workshop's part numbering (①②③④). Define a local
% divider macro instead so we don't touch the shared style file.
% \lessondivider は "Lesson N" という文言を固定で表示するため、①②③④という
% 本講座の区切りには合わない。共有スタイルファイルを変更せずに済むよう、
% ローカルに専用の区切りマクロを定義する。
\newcommand{\dxhpart}[2]{%
  \begin{frame}[plain]
    \centering
    \vspace*{\fill}
    \begin{tikzpicture}
      \node[
        rectangle, rounded corners=8pt,
        draw=sfblue, line width=1.5pt,
        fill=sflight,
        inner xsep=20pt, inner ysep=16pt,
        align=center,
        text width=115mm,
      ] {%
        {\fontsize{34}{40}\selectfont\bfseries\textcolor{sfblue}{#1}}\\[8pt]
        {\Large\bfseries\textcolor{sfdark}{#2}}%
      };
    \end{tikzpicture}
    \vspace*{\fill}
    \par
    {\small\textcolor{sfgray}{StampFly DXH ワークショップ}}\par
    \vspace{8mm}
  \end{frame}
}
